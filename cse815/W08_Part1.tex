\section{Using Per-Item Features}

\begin{frame}{If we know what each dish is like \dots}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}lcc@{}}
        \toprule
        dish & $x_1$ spicy & $x_2$ sweet\\
        \midrule
        Chicken Curry          & $0.90$ & $0$\\
        Chilli Noodles       & $1.00$ & $0.01$\\
        Pepper Soup  & $0.99$ & $0$\\
        Ginger Cake    & $0.10$ & $1.00$\\
        Mango Ice Cream    & $0$    & $0.90$\\
        \bottomrule
      \end{tabular}\par
      \vspace{2mm}
      \begin{block}{A linear model per user}
        Predict user $j$'s rating of dish $i$ as
        \[ \vw^{(j)}\cdot\vx^{(i)} + b^{(j)} . \]
        Each user has their own $\vw^{(j)}, b^{(j)}$: \emph{linear regression,
        once per user}, on the dishes that user rated.
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
      \begin{block}{Worked example}
        Suppose Alice's parameters are $\vw^{(1)} = (5,\ 0)$, $b^{(1)} = 0$.
        For \emph{Pepper Soup}, $\vx^{(3)} = (0.99,\ 0)$:
        \[ \vw^{(1)}\cdot\vx^{(3)} + b^{(1)} = 5(0.99) + 0(0) + 0 = \mathbf{4.95} \]
        --- recommend it.
      \end{block}
      \begin{defnbox}[Cost for user $j$]
        \small
        \[ \tfrac12\!\!\sum_{i:\,r(i,j)=1}\!\!\bigl(\vw^{(j)}\cdot\vx^{(i)} + b^{(j)} - y^{(i,j)}\bigr)^2
           + \tfrac{\lambda}{2}\sum_k \bigl(w^{(j)}_k\bigr)^2 \]
        Sum over users to learn all $\vw^{(j)}, b^{(j)}$ at once.
      \end{defnbox}
    \end{column}
  \end{columns}
\end{frame}

\section{Collaborative Filtering}

\begin{frame}{Turn the problem around}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{If we know the users \dots}
        \footnotesize
        Suppose we knew every user's $\vw^{(j)}, b^{(j)}$. Then for each dish $i$ we could
        \emph{learn} its features $\vx^{(i)}$ from the ratings it received:
        \[ \min_{\vx^{(i)}}\ \tfrac12\!\!\sum_{j:\,r(i,j)=1}\!\!\bigl(\vw^{(j)}\cdot\vx^{(i)} + b^{(j)} - y^{(i,j)}\bigr)^2
           + \tfrac{\lambda}{2}\sum_k \bigl(x^{(i)}_k\bigr)^2 \]
        Possible only because \emph{many} users rated each dish: their ratings
        collectively say what the dish is like.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{defnbox}[Collaborative filtering cost]
        \small
        Learn $\vw$, $b$ \emph{and} $\vx$ together:
        \begin{align*}
          J &= \tfrac12\!\!\sum_{(i,j):\,r(i,j)=1}\!\!\bigl(\vw^{(j)}\cdot\vx^{(i)} + b^{(j)} - y^{(i,j)}\bigr)^2\\
            &\quad + \tfrac{\lambda}{2}\sum_{j,k}\bigl(w^{(j)}_k\bigr)^2
               + \tfrac{\lambda}{2}\sum_{i,k}\bigl(x^{(i)}_k\bigr)^2
        \end{align*}
      \end{defnbox}
      \footnotesize
      \textbf{Gradient descent} on all three:
      \[ w^{(j)}_k \mathrel{-}= \alpha\tfrac{\partial J}{\partial w^{(j)}_k},\quad
         b^{(j)} \mathrel{-}= \alpha\tfrac{\partial J}{\partial b^{(j)}},\quad
         x^{(i)}_k \mathrel{-}= \alpha\tfrac{\partial J}{\partial x^{(i)}_k} \]
      ``Collaborative'': every user's ratings help learn every dish's features, which
      help predict every other user's ratings.
    \end{column}
  \end{columns}
\end{frame}

\section{Binary Labels: Likes, Favourites and Clicks}

\begin{frame}{Many systems never ask for a rating}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{Implicit, binary signals}
        \footnotesize
        \begin{itemize}
          \item Did the user buy the item after seeing it? ($1$ / $0$)
          \item Did they like, favourite or share the post?
          \item Did they order the same dish again within a month?
          \item Did they click the item when shown?
        \end{itemize}
        $1$ = engaged, $0$ = shown but not engaged, $?$ = never shown.
      \end{block}
      \begin{block}{From regression to classification}
        \footnotesize Exactly the step from Week 1 to Week 3:
        \[ P\bigl(y^{(i,j)} = 1\bigr) = g\bigl(\vw^{(j)}\cdot\vx^{(i)} + b^{(j)}\bigr),
           \quad g(z)=\tfrac{1}{1+e^{-z}} \]
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{defnbox}[Binary collaborative filtering cost]
        \small
        \[ J = \!\!\sum_{(i,j):\,r(i,j)=1}\!\! L\bigl(f^{(i,j)},\,y^{(i,j)}\bigr)
           + \tfrac{\lambda}{2}\|\mathbf{W}\|^2 + \tfrac{\lambda}{2}\|\mathbf{X}\|^2 \]
        with the binary cross-entropy
        $L(f,y) = -y\ln f - (1-y)\ln(1-f)$.
      \end{defnbox}
      \begin{pitfall}[A ``0'' is not a dislike]
        \footnotesize ``Shown but not clicked'' may mean the user never noticed it. And
        items never shown are \emph{missing}, not $0$: treating them as dislikes teaches
        the model that whatever it did not recommend is bad.
      \end{pitfall}
    \end{column}
  \end{columns}
\end{frame}

\section{Mean Normalization}

\begin{frame}{A new user, Eve, has rated nothing}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{block}{Without normalization}
        Eve's terms appear only in the penalty $\frac{\lambda}{2}\|\vw^{(5)}\|^2$, so
        gradient descent drives $\vw^{(5)} \to \mathbf{0}$ (here $(-0.007,\ -0.002)$), and
        $b^{(5)}$ never moves from $0$. Every prediction for Eve is $\approx 0$:
        \[ (-0.01,\ -0.01,\ -0.00,\ 0.01,\ 0.01) \]
        The system thinks Eve hates everything --- the worst possible start.
      \end{block}
      \begin{block}{With normalization}
        Subtract each dish's mean rating $\mu_i$ (over users who rated it) before
        training; add it back when predicting:
        \[ \text{prediction} = \vw^{(j)}\cdot\vx^{(i)} + b^{(j)} + \mu_i . \]
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.18}
      \begin{tabular}{@{}lrrr@{}}
        \toprule
        dish & $\mu_i$ & Eve, raw & Eve, normalized\\
        \midrule
        Curry & $2.50$ & $-0.01$ & $2.49$\\
        Noodles & $2.50$ & $-0.01$ & $2.49$\\
        Soup & $2.00$ & $-0.00$ & $1.99$\\
        Ginger Cake & $2.25$ & $0.01$ & $2.26$\\
        Ice Cream & $1.67$ & $0.01$ & $1.68$\\
        \bottomrule
      \end{tabular}
      \vspace{2mm}
      \begin{keyidea}
        With normalization a new user is predicted the \emph{average} rating of each
        dish --- a sensible default until their own ratings arrive. It also helps
        training converge faster.
      \end{keyidea}
    \end{column}
  \end{columns}
\end{frame}

\section{TensorFlow Implementation}

\begin{frame}[fragile]{Automatic differentiation does the calculus}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
\begin{lstlisting}
X = tf.Variable(rng.normal(0, .5, (n_m, n)))
W = tf.Variable(rng.normal(0, .5, (n_u, n)))
b = tf.Variable(tf.zeros((1, n_u)))
opt = keras.optimizers.Adam(learning_rate=0.1)

for it in range(200):
    with tf.GradientTape() as tape:
        P = tf.matmul(X, W, transpose_b=True) + b
        E = (P - Ynorm) * R          # only rated
        J = 0.5 * tf.reduce_sum(E ** 2) \
            + lam / 2 * (tf.reduce_sum(X ** 2)
                         + tf.reduce_sum(W ** 2))
    grads = tape.gradient(J, [X, W, b])
    opt.apply_gradients(zip(grads, [X, W, b]))
\end{lstlisting}
    \end{column}
    \begin{column}{0.4\textwidth}
      \footnotesize
      \begin{block}{Why a custom loop?}
        Collaborative filtering is not a \texttt{Sequential} stack of layers: the
        ``inputs'' $\vx^{(i)}$ are themselves parameters. So we write the cost, and let
        \texttt{GradientTape} compute every partial derivative (Week 5's
        back-propagation, automated).
      \end{block}
      \begin{keyidea}[Vectorized]
        $\mathbf{X}\mathbf{W}^{\mathsf T} + \mathbf{b}$ predicts the whole $n_m\times n_u$
        matrix at once; multiplying by $\mathbf{R}$ zeroes the unrated entries so they
        contribute nothing.
      \end{keyidea}
    \end{column}
  \end{columns}
\end{frame}

\section{Wrap-up and Assessment}

\begin{frame}{Summary}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{What to remember}
        \footnotesize
        \begin{itemize}
          \item The ratings matrix: $n_m$ items $\times$ $n_u$ users, mostly missing.
          \item Content-based: known item features, one linear regression per user.
          \item Collaborative filtering learns item features and user parameters
                together from the ratings alone.
          \item Binary signals (likes, clicks) use the logistic version.
          \item Mean normalization gives new users the average prediction.
          \item Implementation: write the cost, let \texttt{GradientTape} differentiate.
        \end{itemize}
      \end{block}

        \[ J = \tfrac12\!\!\sum_{r(i,j)=1}\!\!(\hat y^{(i,j)} - y^{(i,j)})^2
               + \tfrac{\lambda}{2}(\|\mathbf W\|^2 + \|\mathbf X\|^2) \]
      \end{block}

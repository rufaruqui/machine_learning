      \begin{defnbox}[Clustering]
        \small Given unlabelled data, find groups --- \textbf{clusters} --- of points that are
        similar to each other and different from points in other groups.
      \end{defnbox}
      \begin{block}{Where it is used}
        \begin{itemize}
          \item Grouping similar news articles.
          \item Market segmentation: customers who buy alike.
          \item DNA analysis; grouping stars into galaxies.
        \end{itemize}
      \end{block}
      \begin{keyidea}
        You see three groups at a glance. K-means finds them --- in any number of

\section{The K-means Algorithm}

\begin{frame}{K-means, precisely}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{defnbox}[K-means]
        \small
        Randomly initialise $K$ cluster centroids $\mu_1,\dots,\mu_K\in\R^n$.\\[2pt]
        Repeat \{\\[1pt]
        \hspace*{1em}\textbf{for} $i=1$ to $m$:\\
        \hspace*{2em}$c^{(i)}:=$ index $k$ of the centroid closest to $\vx^{(i)}$,\\
        \hspace*{2em}i.e.\ $\displaystyle c^{(i)}=\argmin_k \|\vx^{(i)}-\mu_k\|^2$\\[2pt]
        \hspace*{1em}\textbf{for} $k=1$ to $K$:\\
        \hspace*{2em}$\mu_k:=$ mean of the points assigned to cluster $k$\\
        \}
      \end{defnbox}
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
      \begin{block}{Notation}
        $c^{(i)}\in\{1,\dots,K\}$: the cluster of example $i$.\\
        $\mu_k\in\R^n$: the centroid of cluster $k$ --- a point in feature space, not a
        training example.\\
        $\mu_{c^{(i)}}$: the centroid of the cluster that $\vx^{(i)}$ is in.
      \end{block}
      \begin{pitfall}[An empty cluster]
        \footnotesize If no point is assigned to some $\mu_k$, its mean is $0/0$. Either
        drop that cluster (continue with $K-1$) or re-initialise it at a random example.
        In our data, starting all three centroids bunched in one corner produces exactly
        this.
      \end{pitfall}
      Squared distance is used because it is cheaper (no square root) and selects the same
      nearest centroid.
    \end{column}
  \end{columns}
\end{frame}

      \begin{defnbox}[Distortion (cost function)]
        \small
        \[ J\bigl(c^{(1)},\dots,c^{(m)},\mu_1,\dots,\mu_K\bigr)
           =\frac1m\sum_{i=1}^{m}\bigl\|\vx^{(i)}-\mu_{c^{(i)}}\bigr\|^2 \]
        The average squared distance from each point to the centroid of its cluster.
      \end{defnbox}
      \footnotesize
      K-means is \textbf{coordinate descent} on $J$: it minimises over one group of
      variables while holding the other fixed, and alternates.
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{Why neither step can increase $J$}
        \textbf{Assign} (hold $\mu$ fixed, choose $c$): each term
        $\|\vx^{(i)}-\mu_{c^{(i)}}\|^2$ is made as small as possible by the nearest
        centroid.\\[3pt]
        \textbf{Move} (hold $c$ fixed, choose $\mu_k$): $\sum_{i:c^{(i)}=k}\|\vx^{(i)}-\mu_k\|^2$
        is minimised by the mean. Setting its gradient
        $-2\sum_{i:c^{(i)}=k}(\vx^{(i)}-\mu_k)$ to zero gives exactly
        $\mu_k=\text{mean}$.
      \end{block}
      \begin{keyidea}[Consequences]
        $J$ decreases or stays the same at every step, and there are finitely many
        assignments, so K-means always converges. If $J$ ever rises, the code has a bug.
      \end{keyidea}
      Hand example: $J=3.464\to3.292\to\dots\to1.218$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Multiple random initialisations}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{defnbox}[Recipe]
        \small
        \textbf{for} $t=1$ to $100$ (typically $50$--$1000$):\\
        \hspace*{1em}randomly initialise K-means;\\
        \hspace*{1em}run it to convergence, giving $c^{(i)}$, $\mu_k$;\\
        \hspace*{1em}compute the distortion $J$.\\
        Keep the run with the \textbf{lowest} $J$.
      \end{defnbox}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{How many runs are enough?}
        If one run finds the best clustering with probability $p$, then $t$ runs all miss it
        with probability $(1-p)^t$. Here $p=193/200=0.965$: two runs miss with probability
        $0.035^2\approx0.0012$.
      \end{block}
      \begin{keyidea}
        Comparing runs needs a number, and $J$ is that number --- one more use of having
        an explicit objective.
      \end{keyidea}
      \begin{pitfall}[Harder problems]
        \footnotesize With many clusters or high dimensions $p$ can be small; that is when
        hundreds of restarts, or smarter seeding such as k-means++, pay off.
      \end{pitfall}
    \end{column}
  \end{columns}
\end{frame}

        $J$ falls steeply up to $K=3$ ($14.41\to8.13\to2.34$) and slowly after
        ($1.92$, $1.52$, \dots). The bend --- the ``elbow'' --- suggests $K=3$.
      \end{block}
      \begin{pitfall}[Do not choose $K$ by minimising $J$]
        \footnotesize $J$ always decreases as $K$ grows ($K=m$ gives $J=0$). And many real
        data sets have no clear elbow: the curve bends smoothly.
      \end{pitfall}
    \end{column}

\begin{frame}{Choose $K$ for the purpose}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{T-shirt sizes}
        \footnotesize
        Cluster customers by height and weight. $K=3$ gives S, M, L; $K=5$ gives XS, S,
        M, L, XL. Neither is ``correct''. Five sizes fit customers better; three are cheaper
        to manufacture and stock. The business decides.
      \end{block}
      \begin{keyidea}
        Clustering is usually a step towards something else. Evaluate $K$ by how well the
        clusters serve that later purpose.
      \end{keyidea}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{Other practical uses}
        \begin{itemize}
          \item \textbf{Image compression}: cluster pixel colours into $K=16$ colours and
                store one 4-bit index per pixel instead of 24 bits.
          \item \textbf{Pre-processing}: use cluster membership as a feature for a
                supervised model.
        \end{itemize}
      \end{block}
      \begin{block}{Questions to ask}
        How many groups can the user act on? What does one more cluster cost? Do the
        clusters stay stable across random restarts?
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Wrap-up and Assessment}

\begin{frame}{Summary}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{block}{What to remember}
        \footnotesize
        \begin{itemize}
          \item Clustering finds groups in unlabelled data.
          \item K-means alternates: assign each point to its nearest centroid; move each
                centroid to its cluster's mean.
          \item It is coordinate descent on the distortion $J$, which never increases.
          \item It converges, but possibly to a local minimum: run many random
                initialisations and keep the lowest $J$.
          \item Choose $K$ by the elbow if there is one --- and by the downstream purpose
                in any case.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Formulas}
        \[ c^{(i)}=\argmin_k\|\vx^{(i)}-\mu_k\|^2 \]
        \[ \mu_k=\frac{1}{|C_k|}\sum_{i\in C_k}\vx^{(i)} \]
        \[ J=\frac1m\sum_i\|\vx^{(i)}-\mu_{c^{(i)}}\|^2 \]
      \end{block}

\begin{frame}{Design questions}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Design a segmentation.} A Chattogram supermarket chain has a year of
          purchase records for $200\,000$ loyalty-card holders. Design a K-means
          segmentation: the features (and their scaling), how you choose $K$, and how
          marketing will use the clusters. \hfill\clo{12}
    \item \textbf{Design for scale.} K-means on $10^7$ points with $K=100$ is too slow to
          restart $500$ times. Design a cheaper procedure that still guards against poor
          local minima, and say how you would check it. \hfill\clo{10}
    \item \textbf{Design a test of a K-means implementation.} Design three small data sets
          with known answers that would expose: a wrong assignment rule, a wrong centroid
          update, and missing empty-cluster handling. \hfill\clo{10}
    \item \textbf{Design a stability check.} Design an experiment that tells a client
          whether the clusters you report are real structure or an artefact of the random
          initialisation. \hfill\clo{12}
  \end{enumerate}
